import java.util.Scanner;

public class ListaCompras {
    public static void main(String[] args) {

        Scanner entrada = new Scanner(System.in);

        System.out.print("Quantos itens deseja adicionar? ");
        int quantidade = entrada.nextInt();

        entrada.nextLine();

        String[] itens = new String[quantidade];
        double[] precos = new double[quantidade];

        double total = 0;

        for (int i = 0; i < quantidade; i++) {

            System.out.print("\nDigite o nome do item " + (i + 1) + ": ");
            itens[i] = entrada.nextLine();

            System.out.print("Digite o preço: R$ ");
            precos[i] = entrada.nextDouble();

            entrada.nextLine();

            total = total + precos[i];
        }

        System.out.println("\n--- RESUMO DA COMPRA ---");

        for (int i = 0; i < quantidade; i++) {
            System.out.printf("%d - %s: R$ %.2f%n",
                    i + 1, itens[i], precos[i]);
        }

        System.out.println("------------------------");
        System.out.printf("TOTAL: R$ %.2f%n", total);

        entrada.close();
    }
}
